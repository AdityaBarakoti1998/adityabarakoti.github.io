---- Q11: Check if a given number is Armstrong ----
import java.util.*;
public class Q11 {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        System.out.print("Enter number: ");
        int n = sc.nextInt();
        int original = n, digits = String.valueOf(n).length(), sum = 0;
        while (n != 0) {
            int d = n % 10;
            sum += Math.pow(d, digits);
            n /= 10;
        }
        System.out.println(original == sum ? "Armstrong Number" : "Not Armstrong Number");
    }
}

---- Q22: Check if a given string is a palindrome using a loop ----
import java.util.*;
public class Q22 {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        System.out.print("Enter string: ");
        String s = sc.nextLine().toLowerCase();
        int left = 0, right = s.length() - 1;
        boolean isPalindrome = true;
        while (left < right) {
            if (s.charAt(left) != s.charAt(right)) {
                isPalindrome = false;
                break;
            }
            left++;
            right--;
        }
        System.out.println(isPalindrome ? "Palindrome" : "Not Palindrome");
    }
}

---- Q29: Check if a given number is a strong number ----
// Strong number: sum of factorial of digits == the number itself (e.g. 145)
import java.util.*;
public class Q29 {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        System.out.print("Enter number: ");
        int n = sc.nextInt();
        int original = n, sum = 0;
        while (n != 0) {
            int d = n % 10;
            long fact = 1;
            for (int i = 1; i <= d; i++) fact *= i;
            sum += fact;
            n /= 10;
        }
        System.out.println(sum == original ? "Strong Number" : "Not a Strong Number");
    }
}

---- Q32: Check if a given string is a pangram ----
// Pangram: contains every letter of the alphabet at least once
import java.util.*;
public class Q32 {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        System.out.print("Enter string: ");
        String s = sc.nextLine().toLowerCase();
        boolean[] seen = new boolean[26];
        for (char c : s.toCharArray()) {
            if (c >= 'a' && c <= 'z') seen[c - 'a'] = true;
        }
        boolean isPangram = true;
        for (boolean b : seen) {
            if (!b) { isPangram = false; break; }
        }
        System.out.println(isPangram ? "Pangram" : "Not a Pangram");
    }
}

---- Q37: Pattern printing (8-10 common patterns) ----
public class Q37 {
    public static void main(String[] args) {
        int n = 5;

        // 1. Right-angled triangle
        System.out.println("1. Right-Angled Triangle:");
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= i; j++) System.out.print("* ");
            System.out.println();
        }

        // 2. Inverted right-angled triangle
        System.out.println("\n2. Inverted Right-Angled Triangle:");
        for (int i = n; i >= 1; i--) {
            for (int j = 1; j <= i; j++) System.out.print("* ");
            System.out.println();
        }

        // 3. Pyramid
        System.out.println("\n3. Pyramid:");
        for (int i = 1; i <= n; i++) {
            for (int s = 1; s <= n - i; s++) System.out.print("  ");
            for (int j = 1; j <= (2 * i - 1); j++) System.out.print("* ");
            System.out.println();
        }

        // 4. Inverted Pyramid
        System.out.println("\n4. Inverted Pyramid:");
        for (int i = n; i >= 1; i--) {
            for (int s = 1; s <= n - i; s++) System.out.print("  ");
            for (int j = 1; j <= (2 * i - 1); j++) System.out.print("* ");
            System.out.println();
        }

        // 5. Diamond
        System.out.println("\n5. Diamond:");
        for (int i = 1; i <= n; i++) {
            for (int s = 1; s <= n - i; s++) System.out.print("  ");
            for (int j = 1; j <= (2 * i - 1); j++) System.out.print("* ");
            System.out.println();
        }
        for (int i = n - 1; i >= 1; i--) {
            for (int s = 1; s <= n - i; s++) System.out.print("  ");
            for (int j = 1; j <= (2 * i - 1); j++) System.out.print("* ");
            System.out.println();
        }

        // 6. Hollow Square
        System.out.println("\n6. Hollow Square:");
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= n; j++) {
                if (i == 1 || i == n || j == 1 || j == n) System.out.print("* ");
                else System.out.print("  ");
            }
            System.out.println();
        }

        // 7. Hollow Right-Angled Triangle
        System.out.println("\n7. Hollow Right-Angled Triangle:");
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= i; j++) {
                if (j == 1 || j == i || i == n) System.out.print("* ");
                else System.out.print("  ");
            }
            System.out.println();
        }

        // 8. Hollow Diamond
        System.out.println("\n8. Hollow Diamond:");
        for (int i = 1; i <= n; i++) {
            for (int s = 1; s <= n - i; s++) System.out.print("  ");
            for (int j = 1; j <= (2 * i - 1); j++) {
                if (j == 1 || j == (2 * i - 1)) System.out.print("* ");
                else System.out.print("  ");
            }
            System.out.println();
        }
        for (int i = n - 1; i >= 1; i--) {
            for (int s = 1; s <= n - i; s++) System.out.print("  ");
            for (int j = 1; j <= (2 * i - 1); j++) {
                if (j == 1 || j == (2 * i - 1)) System.out.print("* ");
                else System.out.print("  ");
            }
            System.out.println();
        }

        // 9. Pascal's Triangle (numbers)
        System.out.println("\n9. Pascal's Triangle:");
        for (int i = 0; i < n; i++) {
            int val = 1;
            for (int s = 0; s < n - i; s++) System.out.print("  ");
            for (int j = 0; j <= i; j++) {
                System.out.print(val + " ");
                val = val * (i - j) / (j + 1);
            }
            System.out.println();
        }

        // 10. Number Pyramid
        System.out.println("\n10. Number Pyramid:");
        for (int i = 1; i <= n; i++) {
            for (int s = 1; s <= n - i; s++) System.out.print("  ");
            for (int j = 1; j <= i; j++) System.out.print(j + " ");
            for (int j = i - 1; j >= 1; j--) System.out.print(j + " ");
            System.out.println();
        }
    }
}
